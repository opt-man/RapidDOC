cmake_minimum_required(VERSION 3.16.3)
project(figure8_three_reference LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(F8_ASSETS "${CMAKE_CURRENT_SOURCE_DIR}/assets" CACHE PATH "Generated model")
option(F8_BUILD_OURS "Build Ours" ON)
option(F8_BUILD_CASADI "Build CasADi/IPOPT" ON)
option(F8_BUILD_ACADOS "Build acados" ON)
function(backend name)
  add_library(f8_${name} SHARED src/api.cpp src/model.c src/${name}.cpp "${F8_ASSETS}/model_generated.c")
  target_compile_definitions(f8_${name} PRIVATE F8_BUILD)
  target_compile_options(f8_${name} PRIVATE -O3)
  target_include_directories(f8_${name} PUBLIC include PRIVATE "${F8_ASSETS}" src)
  target_link_libraries(f8_${name} PRIVATE ${CMAKE_DL_LIBS} m)
  add_executable(experiment_${name} src/experiment.cpp)
  target_compile_definitions(experiment_${name} PRIVATE F8_ASSETS_DIRECTORY="${F8_ASSETS}")
  target_include_directories(experiment_${name} PRIVATE "${F8_ASSETS}")
  target_link_libraries(experiment_${name} PRIVATE f8_${name})
endfunction()
if(F8_BUILD_OURS)
  find_package(fatrop 1.1.7 CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  backend(ours)
  target_link_libraries(f8_ours PRIVATE fatrop::fatrop Threads::Threads OpenMP::OpenMP_CXX)
endif()
if(F8_BUILD_CASADI)
  find_package(casadi CONFIG REQUIRED)
  backend(casadi)
  target_link_libraries(f8_casadi PRIVATE casadi::casadi)
endif()
if(F8_BUILD_ACADOS)
  set(ACADOS_SOURCE_DIR "" CACHE PATH "Native acados installation")
  find_path(F8_ACADOS_INCLUDE acados_c/ocp_nlp_interface.h HINTS "${ACADOS_SOURCE_DIR}/include" REQUIRED)
  foreach(lib acados hpipm blasfeo)
    find_library(F8_${lib}_LIBRARY NAMES ${lib} HINTS "${ACADOS_SOURCE_DIR}/lib" NO_DEFAULT_PATH REQUIRED)
  endforeach()
  foreach(model f8_forward f8_sensitivity)
    set(dir "${F8_ASSETS}/${model}")
    file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${dir}/*.c")
    list(FILTER sources EXCLUDE REGEX "(main_|/main\\.|mex|sfunction|acados_sim_solver)")
    add_library(${model}_generated STATIC ${sources})
    target_include_directories(${model}_generated PUBLIC "${dir}" "${F8_ACADOS_INCLUDE}"
      "${F8_ACADOS_INCLUDE}/blasfeo/include" "${F8_ACADOS_INCLUDE}/hpipm/include")
    target_link_libraries(${model}_generated PUBLIC "${F8_acados_LIBRARY}" "${F8_hpipm_LIBRARY}" "${F8_blasfeo_LIBRARY}")
  endforeach()
  backend(acados)
  target_link_libraries(f8_acados PRIVATE f8_forward_generated f8_sensitivity_generated)
endif()
